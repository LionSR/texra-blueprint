\begin{theorem}[Base result]
    \label{thm:base_result}
    \lean{Fixture.baseResult, Fixture.baseResult}
    \leanok
    A label with an underscore inside a display survives:
    \begin{align}
        [A, B] &= AB - BA, \label{eq:comm_bracket}\\[\jot]
        \begin{psmallmatrix} 1 & 0 \\ 0 & 1 \end{psmallmatrix} x &= x.
    \end{align}
    The bracketed commutator after the row break in
    \eqref{eq:comm_bracket} is mathematics, not a length:
    \begin{align}
        c &= d \\
        [P, Q] &= 0.
    \end{align}
\end{theorem}
\begin{proof}\leanok Immediate. \end{proof}
